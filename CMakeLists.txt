cmake_minimum_required(VERSION 3.24)
project(octopus_llm_engine VERSION 0.2.0 LANGUAGES CXX)

option(OCTOPUS_BUILD_TESTS "Build the numerical and integration tests" ON)
option(OCTOPUS_BUILD_SERVER "Build the HTTP server" ON)
option(OCTOPUS_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(OCTOPUS_DEPENDENCY_DIR "" CACHE PATH "Directory containing json.hpp and httplib.h for offline builds")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)

add_library(octopus_engine
    src/graph.cpp src/backend.cpp src/inference.cpp src/transformer.cpp
    src/model_io.cpp src/onnx_loader.cpp)
add_library(octopus::engine ALIAS octopus_engine)
target_include_directories(octopus_engine PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(octopus_engine PRIVATE octopus_json PUBLIC Threads::Threads)
target_compile_definitions(octopus_engine PUBLIC OCTOPUS_VERSION="${PROJECT_VERSION}")

if(MSVC)
    target_compile_options(octopus_engine PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(octopus_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OCTOPUS_ENABLE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "Use Clang or GCC for the sanitizer configuration")
    endif()
    target_compile_options(octopus_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(octopus_engine PUBLIC -fsanitize=address,undefined)
endif()

add_executable(octopus src/main.cpp)
target_link_libraries(octopus PRIVATE octopus::engine octopus_json)
if(OCTOPUS_BUILD_SERVER)
    target_sources(octopus PRIVATE src/server.cpp)
    target_link_libraries(octopus PRIVATE octopus_http octopus_json)
    target_compile_definitions(octopus PRIVATE OCTOPUS_HAS_SERVER=1)
endif()

add_executable(basic_inference examples/basic_inference.cpp)
target_link_libraries(basic_inference PRIVATE octopus::engine)
add_executable(inference_bench benches/inference_bench.cpp)
target_link_libraries(inference_bench PRIVATE octopus::engine)

if(OCTOPUS_BUILD_TESTS)
    enable_testing()
    add_executable(octopus_tests tests/test_engine.cpp)
    target_link_libraries(octopus_tests PRIVATE octopus::engine)
    add_test(NAME engine COMMAND octopus_tests)
    add_test(NAME example COMMAND basic_inference)
    add_test(NAME benchmark_smoke COMMAND inference_bench 3)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME model_cli COMMAND "${Python3_EXECUTABLE}"
        "${PROJECT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:octopus>")
    if(OCTOPUS_BUILD_SERVER)
        add_test(NAME http_api COMMAND "${Python3_EXECUTABLE}"
            "${PROJECT_SOURCE_DIR}/tests/test_http.py" "$<TARGET_FILE:octopus>")
        set_tests_properties(http_api PROPERTIES TIMEOUT 45)
    endif()
    set_tests_properties(engine model_cli PROPERTIES TIMEOUT 45)
endif()
